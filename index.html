clear
clc
close all
 
 
 
 
 
 
% Vector de tiempo
t = -7:0.001:6;
 
% Señal x(t)
xf = @(t) ...
1.*(t>=-5 & t<-3) + ...
2.*(t>=-3 & t<-1) + ...
1.5.*(t>=-1 & t<2) + ...
0.5.*(t>=2 & t<4);
 
x = xf(t);
 
% Graficar
figure
plot(t,x,'b','LineWidth',2)
 
grid on
 
xlabel('Tiempo [s]')
ylabel('Amplitud')
title('Señal x(t)')
 
xlim([-8 8])
ylim([-0.5 2.5])
 
xticks(-8:1:8)
yticks(-0.5:0.5:2.5)
 
 
 
%% Punto 2: x(t+3)
 
% Nuevo vector de tiempo para visualizar toda la señal desplazada
t2 = -10:0.001:3;
 
x2 = xf(t2+3);
 
figure
plot(t2,x2,'r','LineWidth',2)
 
grid on
 
xlabel('Tiempo [s]')
ylabel('Amplitud')
title('Señal x(t+3)')
 
xlim([-10 3])
ylim([-0.5 2.5])
 
xticks(-10:1:4)
yticks(-0.5:0.5:2.5)
 
 
 
%% Punto 3: x(-t)
 
t3 = -6:0.001:7;
 
x3 = xf(-t3);
 
figure
plot(t3,x3,'g','LineWidth',2)
 
grid on
 
xlabel('Tiempo [s]')
ylabel('Amplitud')
title('Señal x(-t)')
 
xlim([-8 8])
ylim([-0.5 2.5])
 
xticks(-8:1:8)
yticks(-0.5:0.5:2.5)
 
 
 
%% Punto 4: x(t/3)
 
t4 = -21:0.001:18;
 
x4 = xf(t4/3);
 
figure
plot(t4,x4,'m','LineWidth',2)
 
grid on
 
xlabel('Tiempo [s]')
ylabel('Amplitud')
title('Señal x(t/3)')
 
xlim([-21 18])
ylim([-0.5 2.5])
 
xticks(-22:2:19)
yticks(-0.5:0.5:2.5)
 
 
 
%% Punto 5: x(-2t+4)
 
t5 = -1:0.001:5.5;
 
x5 = xf(-2*t5 + 4);
 
figure
plot(t5,x5,'y','LineWidth',2)
 
grid on
 
xlabel('Tiempo [s]')
ylabel('Amplitud')
title('Señal x(-2t+4)')
 
xlim([-2 6])
ylim([-0.5 2.5])
 
xticks(-2:1:6)
yticks(-0.5:0.5:2.5)
 
 
 
%% Punto 6
 
t6 = -14:0.001:9;
 
% Escalón unitario
u1 = @(t) 1.*(t>0);
 
% Ventana
ventana = u1(t6+2) - u1(t6-1);
 
% Señal transformada
x6 = xf((-3/5)*t6 - 2);
 
% Señal final
y = x6.*ventana;
 
figure
plot(t6,y,'b','LineWidth',2)
 
grid on
 
xlabel('Tiempo [s]')
ylabel('Amplitud')
title('y(t)=x(-3t/5-2)[u(t+2)-u(t-1)]')
 
xlim([-14 9])
ylim([-0.5 2.5])
 
xticks(-14:1:9)
yticks(-0.5:0.5:2.5)
